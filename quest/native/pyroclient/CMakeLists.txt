cmake_minimum_required(VERSION 3.22)
project(vibertemis_pyroclient LANGUAGES CXX)
if(NOT ANDROID)
    message(FATAL_ERROR "pyroclient requires Android Vulkan/AHardwareBuffer")
endif()
if(NOT PYROWAVE_SOURCE OR NOT PYROWAVE_LIBRARY)
    message(FATAL_ERROR "Pinned PYROWAVE_SOURCE and built PYROWAVE_LIBRARY required")
endif()
add_library(pyroclient SHARED pyroclient.cpp)
target_compile_features(pyroclient PRIVATE cxx_std_17)
target_compile_options(pyroclient PRIVATE -Wall -Wextra -Wno-missing-field-initializers)
target_include_directories(pyroclient PRIVATE "${PYROWAVE_SOURCE}" "${PYROWAVE_SOURCE}/Granite/third_party/khronos/vulkan-headers/include")
add_library(pyrowave SHARED IMPORTED)
set_target_properties(pyrowave PROPERTIES IMPORTED_LOCATION "${PYROWAVE_LIBRARY}")
target_link_libraries(pyroclient PRIVATE pyrowave vulkan android log)
target_link_options(pyroclient PRIVATE "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map")
set_property(TARGET pyroclient APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/exports.map")
